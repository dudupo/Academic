
\subsection{The Intermediate Argument} 
An essential ingredient of our proof is an 'intermediate argument', stating that prior to having a large slice that closes a non-trivial loop on the Toric, there must be a step in which there is a slice that on one hand doesn't close a non-trivial loop, yet it has a size that is comparable to the Toric side. Before the first step of that kind, we can assume that no slice closes a non-trivial loop; namely, the slices are finite. That implies the following two: first, the condition for \Cref{claim:decodestopnoise} is valid, and therefore the replacement argument can be used to infer a correctness of logical measurement gadgets. Second, the sweep rule is, in some sense, effective (\Cref{claim:polesmove}), and soon we will see that it sets a structure for the histories of slices. 

    \begin{claim}
Consider a chain $y \in \mathbb{F}_{2}^{\mathcal{F}_{d^{\prime}}}$, and let $x_{1}, \ldots, x_{m} \in \mathbb{F}_{2}^{\mathcal{F}_{d^{\prime}}}$ be $d^{\prime}$-dimensional chains such that $y = x_{1} \cup x_{2} \cup \ldots \cup x_{m}$, $y$ is connected, and there is a selection of reduced forms for the $x_{i}$ such that the chain:
      \begin{equation*}
        \label{equ:cond}
        \begin{split}
          \text{reduced form of }(x_{1}) \cup \text{reduced form of }(x_{2}) \cup ,.., \text{reduced form of }(x_{m}) \quad \quad \quad (\star)
        \end{split}
      \end{equation*}
is connected. Then: 
      \begin{equation*}
        \begin{split}
          \Size{ \text{reduced form of } (y) } \le \sum_{i}^{m}{\Size{ \text{reduced form of } (x_{i}) } }
        \end{split}
      \end{equation*}
    \end{claim}
    \begin{proof}
      For each $x_{i}$, denote by $z_{i}$ the stabilizer for which the translation of $x_{i}$ by $z_{i}$ yields the reduced form which satisfies the condition in $(\star)$ . Namely:
      \begin{equation*}
        \begin{split}
          z_{i} = \arg \min_{z \in \mathcal{C}_{Z}^{\perp} }\Size{ x_{i} + z } 
        \end{split}
      \end{equation*}
      Thus:
      \begin{equation*}
        \begin{split}
          \Size{ \text{reduced form of } {y} } & =  \min_{z \in \mathcal{C}_{Z}^{\perp} } \Size{z + y }  \le  \min_{z \in \mathcal{C}_{Z}^{\perp} }\Size{z + x_{1} \cup x_{2} \cup ,.., x_{m}}\\
          & \le \Size{ (x_{1}+z_{1}) \cup (x_{2}+z_{2}) \cup ,.., (x_{m} + z_{i}) } \\
          &\le \sum_{i}^{m}{ \Size{x_{i}+z_{i} } }
        \end{split}
      \end{equation*}
Where the last transition follows from \Cref{claim:tringinq}, combined with the connectivity condition.
    \end{proof}

\begin{claim}
  \label{claim:intersyn}
  Let $\alpha = \frac{1}{2}(\Delta 4^{d})^{-1}$. Denote by $A$ the event that there is a step that admits an infinite slice. In addition, for any step $\tau \in J$, denote by $B_{\tau}$ the event that step $\tau$ is the first step to admit a finite slice at size at least $\alpha n$. Denote by $w$ and $\mu$ the width and the depth of $C[\mathcal{E}]$. Then, for sufficiently large $n$, the probability that $A$ occurs is bounded by:
  \begin{equation*}
    \begin{split}
      \prb{A} \le w\mu\max_{\tau\in J }\prb{B_{\tau}}.
    \end{split}
  \end{equation*}
\end{claim}

\begin{proof}
Assume that turn $\tau+1$ admits an infinite error, and denote by $g$ the gate applied at turn $\tau+1$. Since $\tau+1$ admits an infinite error, there is a reduced Pauli-path $\tilde{P}_{1},..,\tilde{P}_{t-1},\tilde{P}_{t},\tilde{P}_{\tau+1}$ for $C[\mathcal{E}]$ such that $\tilde{P}_{\tau+1}$ is infinite, namely if $x = \text{supp } X(\tilde{P}_{\tau+1})$ and $z = \text{supp } Z(\tilde{P}_{\tau+1})$, then either $x$ or $\phi(z)$ is infinite. Assume that $x$ is infinite, and consider the application of a copy of $g$ in the $\infty$-extension over that aligns with the original gate (the trivial translation). Since $g$ acts on at most $\Delta$ qubits, then it can connect at most $\Delta d$ chains in $\text{supp } X(\tilde{P}_{t})$.


We got that a condition for step $\tau+1$ to admit an infinite slice is that a step $\tau$ admits a slice at size at least $\alpha n$. Thus, the desired inequality follows by taking the union bound:
  \begin{equation*}
    \begin{split}
      \prb{A} \le w\mu\max_{\tau\in J}\prb{B_{\tau}}.
    \end{split}
  \end{equation*}
\end{proof}
