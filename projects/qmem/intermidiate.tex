The following definition of propagation describes the evolution of a fault. It assumes an order of steps and defines an operator that is obtained by commuting a given fault with the gates (and faults) prior to a given step.

\begin{definition}[Fault propagation to a step $\tau$]
  \label{def:localprop}
  Let $C=\{u_i\}_i$ be a quantum circuit, let $J=(j_1,\ldots,j_k)$ be an order agreeing with $\pi_C$, and let $u_i\in C$. Suppose that $u_i=u_{j_l}$ in this order. If $l>\tau$, we define the propagation of $u_i$ to step $\tau$ to be the identity operator.
Otherwise, define operators $\mathcal{D}_0,\mathcal{D}_1,\ldots,\mathcal{D}_{\tau-l}$ recursively by setting $\mathcal{D}_0=u_{j_l}$ and requiring that for every $1\le s\le \tau-l$,
  \begin{equation*}
    \begin{split}
      \mathcal{D}_{s-1}u_{j_{l+s}} = u_{j_{l+s}}\mathcal{D}_s.
    \end{split}
  \end{equation*}
  We then define the \emph{propagation of $u_i$ to step $\tau$} to be the operator
  \begin{equation*}
    \begin{split}
      \mathcal{D} \colonequals \mathcal{D}_{\tau-l}.
    \end{split}
  \end{equation*}
Now let $C[\mathcal{E}]$ be a noisy circuit, where all of its quantum registers are encoded in the $(d,d^{\prime})$-Toric code with side length $n$. Let $J$ be an order agreeing with $\pi_{C[\mathcal{E}]}$, and let $\tau$ be a step in that order. For each faulty gate $e \in \mathcal{E}$ that occurs no later than step $\tau$, denote by $\mathcal{D}^{\tau}_{e}$ its propagation to step $\tau$. We define the propagated error at step $\tau$ by multiplying these propagated faults in reverse topological order.
  \begin{equation*}
    \begin{split}
      \mathcal{D}^{\tau} \colonequals \prod_{e\preceq \tau} \mathcal{D}^{\tau}_{e}.
    \end{split}
  \end{equation*}
Moreover, we say that a step $\tau$ admits an infinite $X$-error if there is a register such that in the view of the $\infty$-extension of $C[\mathcal{E}]$, the propagated error at step $\tau$, $\mathcal{D}^{\tau}$, has support on a connected, reduced, $d^{\prime}$-dimensional chain $x$ such that $\Size{x} \ge n$.
\end{definition}

\end{proof}

\begin{definition}[Deviation clusters configuration]
  \label{def:deviation}
  Let $C$ be a quantum circuit, whose all register are encoded in the $(d,d^{\prime})$-Toric code. Let $R$ be a register, and let $\mathcal{E}$ be a fault set. For a step $\tau$,  let $\mathcal{D}^{\tau}$ be the propagated error at step $\tau$. Let $x \in \mathbb{F}_{2}^{\mathcal{F}_{d^{\prime}}} $ be the chain restricted to a register $R$, and the $X$-component of $\mathcal{D}^{\tau}$. We say that $y \in \mathbb{F}_{2}^{\mathcal{F}_{d^{\prime}}} $ is a  $X$-cluster deviation at step $\tau$, if and only if:
  \begin{enumerate}
    \item There exists $z \in \mathcal{C}_{Z}^{\perp}$ such that $y + z \in x$, namely the faces supporting $y+z$ are contained in the support of $x$.
    \item $y$ is connected component, when two faces are said to be connected if they are share a $d^{\prime}-1$ dimensional face (check).
    \item For any $z \in \mathcal{C}_{Z}^{\perp}$, it holds that $\Size{y} \le \Size{y+z}$. 
    \end{enumerate}
\end{definition}

\begin{claim}
  \label{claim:deviation_clusters}
  Assume that the gates in $C$ have a fan at most $\Delta$. Fix an order $J$ agreeing with $\pi_{C[\mathcal{E}]}$. Let $y$ be a $X$-deviation cluster at step $\tau$. Then there exist $X$-deviation cluster $x$ at step $\tau-1$ such that
  \begin{equation*}
    \begin{split}
      \Size{x} &\ge \frac{1}{4\Delta d^{\prime} (d-d^{\prime}+1)}\,\left(\Size{y } - c\Delta \right)
    \end{split}
  \end{equation*}
\end{claim}
\begin{proof}
Let $u$ be the gate applied at step $\tau$. Passing from step $\tau-1$ to step $\tau$ can modify the deviation only on qubits touched by $u$. Denote by $\xi \in \mathbb{F}_{2}^{\mathcal{F}_{d^{\prime}}}$ the chain corresponding to the addition to the deviation by $u$ restricted to the $X$-component.

Since $u$ acts on at most $\Delta$ qubits, and each such qubit is adjacent to $2d^{\prime}$ checks, each of which meets $2(d-d^{\prime}+1)$ qubits, the gate $u$ can interact with qubits belonging to at most $4\Delta d^{\prime} (d-d^{\prime}+1)$ deviation clusters from step $\tau-1$.

  Let $y_{1},\ldots,y_{m}$, with $m\le 4\Delta d^{\prime} (d-d^{\prime}+1)$, be those deviation clusters at step $\tau-1$, such with $\xi$ 'merged' into $y$. Namely, there is $z \in \mathcal{C}_{Z}^{\perp}$ such: 
  \begin{equation*}
    \begin{split}
        y = \sum_{i=1}^{m}y_{i} + \xi + z 
      \end{split}
  \end{equation*}
  Now, for each $y_{i}$ denote it's restriction to the faces outside the support of $\xi + z$ by $y_{i}|_{/\{\xi + z\}}$. Since $y$ is connected, and for any $i,j$ we have $y_{i} \cap y_{j} = \emptyset$.

  We have that $y_{1}$ and $\sum_{i=2}^{m}y_{i} + \xi + z$ share a check and therefore also share a vertex. Thus:
  \begin{equation*}
    \begin{split}
      \Size{y} = \Size{\sum_{i=1}^{m}y_{i} + \xi + z} \le \Size{y_{1}} + \Size{\sum_{i=2}^{m}y_{i} + \xi + z} 
    \end{split}
  \end{equation*}
By the same argument\footnote{Actually, the order of the exclusion does matter; we take an order such that the remainder remains connected.} we have:
  \begin{equation*}
    \begin{split}
      \Size{y} &\le  \left( \sum_{i=1}^{m-1}\Size{y_{i}} \right) + \Size{y_{m} + z} + \Size{\xi} \\
      & \le \left( \sum_{i=1}^{m}\Size{y_{i}} \right) + \Size{\xi} \\
      & \le 4\Delta d^{\prime} (d-d^{\prime}+1) \max_{i}\Size{y_{i}}  + c \Delta
    \end{split}
  \end{equation*}
  Where we used that $\xi$ is supported on at most $\Delta$ faces. Hence one of the deviation clusters $y_{1},..,y_{m}$, call it $x$, satisfies
  \begin{equation*}
    \begin{split}
      \Size{x} &\ge \frac{1}{4\Delta d^{\prime} (d-d^{\prime}+1)}\,\left(\Size{y } - c\Delta \right)
    \end{split}
  \end{equation*}
\end{proof}

\begin{corollary}[Intermediate Value Theorem]
  \label{claim:inter}
  Let $\alpha = 1/(4\Delta d^{\prime} (d-d^{\prime}+1))$, $\beta = \alpha c\Delta$, and $a$ be an integer. Assume that at the initial step $\tau_{o}$, there is no $X$-deviation cluster at size greater than $\alpha a - \beta$, and that at a step $\tau_{1} \in J$, such that $\tau_{1} > \tau_{o}$, there is a deviation cluster $y$ at size $\Size{y} > a$. Then there is a step $\tau_{\star} \in J$ such that $\tau_{o} < \tau_{\star} < \tau_{1}$ for which there is a deviation cluster $x$ at step $\tau_{\star}$, satisfying that  $\Size{x}$ is in the range $(\alpha a - \beta, a)$.
\end{corollary}
\begin{proof}
  Assume towards contradiction that for every step $\tau$ with $\tau_{o}<\tau<\tau_{1}$, all deviation clusters are at size at most $\alpha a - \beta$. In particular, this holds at step $\tau_{1}-1$. On the other hand, since $\Size{y}>a$, \Cref{claim:deviation_clusters} implies the existence of a deviation cluster at step $\tau_{1}-1$ at size at least
\begin{equation*}
  \begin{split}
    \alpha \Size{x} - \frac{c}{\Delta} > \alpha a - \beta 
  \end{split}
\end{equation*}
which is a contradiction. Therefore, there must exist an intermediate step $\tau_{\star}$ with $\tau_{o}<\tau_{\star}<\tau_{1}$ at which some deviation cluster has a size in the interval $(\alpha a- \beta ,a)$.
\end{proof}

We close the subsection by applying the intermediate theorem to derive a bound on the occurrence of deviation clusters at size greater than $a$. In particular, the intermediate theorem states that for that to happen, there has to be a turn in which there is a deviation cluster that is at size of at least $\alpha a - \beta$.

\subsection{Gottesman argument}

\begin{claim}
  Let $\gamma, \gamma^{\prime} \in (0,1)$, and denote by $n$ the side-length of the Toric. Let $x$ be the error, and assume that the probability of reduced error, time-connected, cluster $x_{i}$ to occur is at most $p^{\Size{x_{i}}}$.  Consider a correction algorithm that applies a correction $f$, such that $\partial^{-}(f + x) = 0$ and $f$ consists of clusters $\{f_{i}\}$ each of size at most $\Size{f_{i}} \le \frac{1}{2}\gamma \sqrt{n}$. Then the algorithm corrects the error.
\end{claim}
\begin{proof}
  
\end{proof}
Suppose, through contradiction, that there is a cluster $y$ in $x+f$ which is not a stabilizer and has a size greater than $\gamma n$. Then $y$ consists of clusters $x_{1}, \ldots, x_{m_{1}}$ and $f_{1}, \ldots, f_{m_{2}}$, such that they are connected. Consider the spanning tree of the systems and group the $x_{i}$'s which are time-connected. Moreover, exclude the $f_{i}$'s which connect $x_{i}$ clusters that are connected in time.


Now, denote by $m$ the remained $x$'s. We have that the probability of the tree to occur is the minimum between $m$ and  
\begin{equation*}
  \begin{split}
    p^{\sum_{i}^{m} \Size{x_{i}}} \le p^{ \Size{y} - \sum_{i}^{m^{\prime}} \Size{f_{i}} } \le  p^{ \gamma n  - m^{\prime} \frac{1}{2}\gamma\sqrt{n} } 
  \end{split}
\end{equation*}

Thus, if $m \ge \sqrt{n}$ then, .., and if lower than $\sqrt{n}$ then.., .

Since, any cluster in $f$ is lower than $\gamma^{\prime}n$ it follows that there is a cluster $x^{\prime}$ in $x$, that has a non-trivial intersection with $f$. Denote, by $f^{\prime}$ the minimal subset of $f$ such that $\partial^{-} (f^{\prime} + x^{\prime}) = 0$. Now, if $f^{\prime} + x^{\prime}$ is stabilizer, then just erase it from $y$, denote by $y^{\prime}$ the result of the erasing (if the erasing divide $y$ to multiple clusters then sets $y^{\prime}$ to the one with the maximal size). Repeat on the above until $f^{\prime}+x^{\prime}$ is not a stabilizer. Therefore, it has to be that $\Size{x^{\prime}+f^{\prime}} > \gamma n$. 
Now, take the components $f_{1},..,f_{d+1}$ such the maximal of $L_{i}$ is in $\Size{x^{\prime}+f^{\prime}}$ is got in $f_{i}$. Since $x^{\prime}+f^{\prime}$ is connected (otherwise $\partial^{-}(x^{\prime}+f^{\prime}) \neq 0$), and $f_{1},..,f_{d+1}$ are disconnected we have that: 


\begin{equation*}
  \begin{split}
 \sum\Size{f_{i}} &> \gamma n - \Size{x^{\prime}} > \gamma^{\prime} n\\
 \Rightarrow & \max \Size{f_{i}} > \frac{\gamma^{\prime}}{d+1} n
  \end{split}
\end{equation*}


\inb{\textbf{Even better.}} Assume that the clusters always have a size lower than $\gamma n$ and that there are $m$ bits. Run sweep-cycles for $m\gamma n$ times. Since there can be at most $m-1$ merges (since the number of clusters is bounded by the number of bits), there has to be a continuous sequence of $\gamma n$ cycles for which no merge occurs. Since it's given that the size of all the clusters is lower than $\gamma n$, we have that all the clusters are erased after those cycles. To prove that no cluster reaches a size greater than $\gamma n$, we use explanation trees (where trees in which all the leaves are at least $l$ from the root are contained in the trees without that restriction).
