The following definition of propagation describes the evolution of a fault. It assumes an order of steps and defines an operator that is obtained by commuting a given fault with the gates (and faults) prior to a given step.

\begin{definition}[Fault propagation to a step $\tau$]
  \label{def:localprop}
  Let $C=\{u_i\}_i$ be a quantum circuit, let $J=(j_1,\ldots,j_k)$ be an order agreeing with $\pi_C$, and let $u_i\in C$. Suppose that $u_i=u_{j_l}$ in this order. If $l>\tau$, we define the propagation of $u_i$ to step $\tau$ to be the identity operator.

  Otherwise, define operators $X_0,X_1,\ldots,X_{\tau-l}$ recursively by setting $X_0=u_{j_l}$ and requiring that for every $1\le s\le \tau-l$,
  \begin{equation*}
    \begin{split}
      X_{s-1}u_{j_{l+s}} = u_{j_{l+s}}X_s.
    \end{split}
  \end{equation*}
  We then define the \emph{propagation of $u_i$ to step $\tau$} to be the operator
  \begin{equation*}
    \begin{split}
      X \colonequals X_{\tau-l}.
    \end{split}
  \end{equation*}

  Now let $C[\mathcal{E}]$ be a noisy circuit, let $J$ be an order agreeing with $\pi_{C[\mathcal{E}]}$, and let $\tau$ be a step in that order. For each fault gate $e\in\mathcal{E}$ that occurs no later than step $\tau$, denote by $X^{\tau}_{e}$ its propagation to step $\tau$. We define the propagated error at step $\tau$ by multiplying these propagated faults in reverse topological order:
  \begin{equation*}
    \begin{split}
      X^{\tau} \colonequals \prod_{e\preceq \tau} X^{\tau}_{e}.
    \end{split}
  \end{equation*}
\end{definition}

The following definitions extend the notion of a Tanner graph to the setting of quantum circuits by joining the Tanner graphs of each encoded register. Deviation clusters are a connected component of the Tanner graph such that each of their checks touches a qubit that belongs to the deviation.

\begin{definition}[Tanner graph of a register]
  Let $C$ be a quantum circuit and let $R$ be a register, namely a subset of qubits of $C$. The \textit{Tanner graph} of $R$ at step $\tau$, denoted by $\mathcal{T}_{R,\tau}$, is defined as follows:
  \begin{enumerate}
    \item If at step $\tau$ the register $R$ is encoded by a code $\mathcal{C}$, then $\mathcal{T}_{R,\tau}$ is the Tanner graph of $\mathcal{C}$.
    \item Otherwise, $\mathcal{T}_{R,\tau}$ is the empty graph.
  \end{enumerate}
  When the step is clear from the context, we omit the subscript $\tau$.
\end{definition}

\begin{definition}[Generalized Tanner graph of a circuit]
  Let $C$ be a quantum circuit. For registers $A$ and $B$ in $C$, define the generalized Tanner graph of their union by
  \begin{equation*}
    \begin{split}
      \mathcal{T}_{A\cup B,\tau} \colonequals \mathcal{T}_{A,\tau}\cup \mathcal{T}_{B,\tau}.
    \end{split}
  \end{equation*}
  More generally, for a partition of the qubits into registers $\{R_i\}_i$, the generalized Tanner graph of $C$ at step $\tau$ is
  \begin{equation*}
    \begin{split}
      \mathcal{T}_{C,\{R_i\},\tau} \colonequals \bigcup_i \mathcal{T}_{R_i,\tau}.
    \end{split}
  \end{equation*}
  When the partition and the step are clear, we abbreviate this notation to $\mathcal{T}_C$.
\end{definition}

\begin{definition}[Clustered deviation]
  Let $C$ be a quantum circuit, let $R$ be a register of $C$, and let $\mathcal{E}$ be a fault set. Let $\mathcal{T}_{R,\tau}$ be the Tanner graph of $R$ at step $\tau$, and let $\mathcal{D}_{\tau}$ be the deviation of $C[\mathcal{E}]$ from $C$ on $R$ at step $\tau$. Denote by $\mathcal{T}_{R,\tau}|_{\mathcal{D}_{\tau}}$ the subgraph induced on the left vertices corresponding to the qubits in $\mathcal{D}_{\tau}$.

  We say that two left vertices are connected if they share a common right neighbor. The \textit{deviation clusters} at step $\tau$ are the connected components of $\mathcal{T}_{R,\tau}|_{\mathcal{D}_{\tau}}$. The size of a deviation cluster is the number of left vertices it contains.


\end{definition}
  
\begin{definition}[Function that respects inclusion]
  \label{def:respinc}
We say that a function from the subsets of bits to the reals $f \colonequals ( \cdot ) \rightarrow \mathbb{R}$ respects inclusion if:
  \begin{enumerate}
    \item It gives zero for the empty set, $f(\emptyset) = 0$.
    \item For any two subsets $A, B$ which are intersecting $A \cap B \neq \emptyset$, we have $f(A \cup B) \le f(A) + f(B)$.
    \item There is a constant $c$ such that the function $f$ on any singleton (a subset containing a single vertex) is bounded by $c$.
  \end{enumerate}

\end{definition}

\begin{claim}
  \label{claim:deviation_clusters}
  Let $C$ be a quantum circuit whose gates act on at most $\Delta$ qubits. Let $R$ be a quantum register encoded by a quantum error-correcting code whose Tanner graph has left degree at most $\Delta_{1}$ and right degree at most $\Delta_{2}$. Fix a fault set $\mathcal{E}$ and an order $J$ agreeing with $\pi_{C[\mathcal{E}]}$. For a step $\tau\in J$, let $y_{1}^{(\tau)},y_{2}^{(\tau)},\ldots$ denote the deviation clusters at step $\tau$.

  Suppose that at step $\tau_{1}$ there is a deviation cluster $y_i^{(\tau_{1})}$. Then there exist deviation cluster $x$  step $\tau_{1}-1$ such that
  \begin{equation*}
    \begin{split}
      x\cap y_i^{(\tau_{1})} \neq \emptyset,
      \qquad
      \Size{x} \ge \frac{1}{\Delta\Delta_{1}\Delta_{2}}\,\Size{y_i^{(\tau_{1})}}- \frac{c}{\Delta}
    \end{split}
  \end{equation*}
\end{claim}
\begin{proof}
  Let $u$ be the gate applied at step $\tau_{1}$. Passing from step $\tau_{1}-1$ to step $\tau_{1}$ can modify the deviation only on qubits touched by $u$. Since $u$ acts on at most $\Delta$ qubits, and each such qubit is adjacent in the Tanner graph to at most $\Delta_{1}$ checks, each of which meets at most $\Delta_{2}$ qubits, the gate $u$ can interact with vertices belonging to at most $\Delta\Delta_{1}\Delta_{2}$ deviation clusters from step $\tau_{1}-1$.


  Let $y_{1},\ldots,y_{m}$, with $m\le \Delta\Delta_{1}\Delta_{2}$, be those deviation clusters at step $\tau_{1}-1$ that meet the neighborhood affected by $u$ and whose union contains $y_i^{(\tau_{1})}$, denote the bits in the neighborhood that belong to $\mathcal{D}_{\tau_{1}}$ by $N$. Then, since for any $i,j \le m$ we have $(y_{i} \cup N) \cap (y_{j} \cup N) \neq \emptyset$ it holds: 
  \begin{equation*}
    \begin{split}
      \Size{ y_i^{(\tau_{1})} }  \le \sum_{j=1}^{m}f(y_j \cup N) \le m\max_{1\le j\le m}f(y_j) +f(N)  \le \Delta\Delta_{1}\Delta_{2}\max_{1\le j\le m}(f(y_j) + c\Delta).
    \end{split}
  \end{equation*}
  Where we used that $N$ is at size at most $\Delta$. Hence one of these clusters, call it $Z$, satisfies
  \begin{equation*}
    \begin{split}
      f(Z) &\ge \frac{1}{\Delta\Delta_{1}\Delta_{2}}\,f(Y_i^{(\tau_{1})}) - \frac{c}{\Delta} 
    \end{split}
  \end{equation*}
  By construction, $Z\cap Y_i^{(\tau_{1})}\neq \emptyset$.
\end{proof}

\begin{corollary}[Intermediate Value Theorem]
  \label{claim:inter}
Let $\alpha = 1/(\Delta \Delta_{1} \Delta_{2})$, $x$ be an integer, and let $f$ be a function that respects inclusion.
  Assume that at the initial step $\tau_{o}$, there is no deviation cluster with an $f$-value greather than $\alpha x - c/\Delta$, and that at a step $\tau_{1} \in J$, such that $\tau_{1} > \tau_{o}$, there is a deviation cluster $Y^{(\tau_{1})}$ with an $f$-value $f(Y^{(\tau_{1})}) > x$. Then there is a step $\tau_{\star} \in J$ such that $\tau_{o} < \tau_{\star} < \tau_{1}$ for which there is a deviation cluster $Z$ at step $\tau_{\star}$, satisfying that  $f(Z)$ is in the range $(\alpha x - c/\Delta, x)$.
\end{corollary}
\begin{proof}
  Assume for contradiction that for every step $\tau$ with $\tau_{o}<\tau<\tau_{1}$, all deviation clusters have an $f$-value at most $\alpha x - c/\Delta$. In particular, this holds at step $\tau_{1}-1$. On the other hand, since $f(Y^{(\tau_{1})})>x$, \Cref{claim:deviation_clusters} implies the existence of a deviation cluster at step $\tau_{1}-1$ with an $f$-value at least
\begin{equation*}
  \begin{split}
    \alpha f(Y^{(\tau_{1})}) - \frac{c}{\Delta} > \alpha x - \frac{c}{\Delta}
  \end{split}
\end{equation*}
which is a contradiction. Therefore, there must exist an intermediate step $\tau_{\star}$ with $\tau_{o}<\tau_{\star}<\tau_{1}$ at which some deviation cluster has an $f$-value in the interval $(\alpha x - c / \Delta ,x)$.
\end{proof}

We close the subsection by applying the intermediate theorem to derive a bound on the occurrence of deviation clusters with an $f$-value greater than $x$. In particular, the intermediate theorem states that for that to happen, there has to be a turn in which there is a deviation cluster that has an $f$-value of at least $(\Delta\Delta_{1}\Delta_{2})^{-1} x - c/\Delta$.

\begin{claim}
  \label{claim:intersyn}
Let $\alpha=(\Delta\Delta_{1}\Delta_{2})^{-1}$. Let $C$ be a quantum circuit with order $J$ and depth $d$, and let $\mathcal{E}\sim\mathcal{N}_{\mathcal{E}}$, so that $C[\mathcal{E}]$ is a random circuit whose gates act on at most $\Delta$ qubits. Fix a time $t\le 2d$ and an integer $x\in\mathbb{N}$. Let $A$ be the event that by time $t$ there exists a deviation cluster with an $f$-value greater than $x$. For each step $\tau\in J$, let $B_{\tau}$ be the event that step $\tau$ is the first step at which there exists a deviation cluster with an $f$-value in the interval $(\alpha x - c/\Delta,x)$. Then
  \begin{equation*}
    \begin{split}
      \prb{A} \le 2wd\max_{\tau\in[2wd]}\prb{B_{\tau}}.
    \end{split}
  \end{equation*}
\end{claim}
\begin{proof}
By \Cref{claim:inter}, whenever the event $A$ occurs there exists a step $\tau\in J$ with $\tau\le 2wt$ at which there is a deviation cluster with an $f$-value in the interval $(\alpha x - c / \Delta,x)$. Hence
  \begin{equation*}
    \begin{split}
      A \subset \bigcup_{\tau\in[2wt]} B_{\tau} \subset \bigcup_{\tau\in[2wd]} B_{\tau}.
    \end{split}
  \end{equation*}
The union bound now gives
  \begin{equation*}
    \begin{split}
      \prb{A}
      \le \prb{\bigcup_{\tau\in[2wd]} B_{\tau}}
      \le \sum_{\tau\in[2wd]} \prb{B_{\tau}}
      \le 2wd\max_{\tau\in[2wd]}\prb{B_{\tau}}.
    \end{split}
  \end{equation*}
\end{proof}




\begin{claim}
  Let $x \in \mathbb{F}_{2}^{\mathcal{F}_{d^{\prime}}}$ be a $d^{\prime}$-dimensional chain, consider the stabilizer  $z \in \mathcal{C}_{Z}^{\perp}$ which adding it to $x$ minimaize the size , Namely: 
  \begin{equation*}
    \begin{split}
      z =  \arg \min_{z \in \mathcal{C}_{Z}^{\perp}} \Size{ x + z }  
    \end{split}
  \end{equation*}
  Then: 
  \begin{equation*}
    \begin{split}
      \Size{ x + z } = \Size{ \partial^{-}( x + z) } 
    \end{split}
  \end{equation*}
\end{claim}
\begin{proof}
  First, observe that since the vertex domain of $\partial^{-} x$ is contained in the vertex domain of $x$, it follows that for any $x$ the inequality $\Size{x} \ge \Size{\partial^{-} x}$ holds. Combining this with the fact that $\partial^{-} z = 0$ for any $z \in \mathcal{C}_{Z}^{\perp}$, we have that $\Size{x + z} \ge \Size{\partial^{-}(x + z)}$. Now suppose that $\Size{x} > \Size{\partial^{-} x}$; that implies the existence of $v \in x/\partial^{-} x$ such that $v$ is a local maximum of one of the potential walls in $x$ and $(\partial^{-} x)|_{v} = 0$. 
 

  If $v$ is a maximum of $L_{d+1}$, then by \Cref{claim:agreement}, there exists a stabilizer $z$ rooted at $v$, such that adding $z$ to $x$ excludes $v$, so $L_{d+1}(x+z) < L_{d+1}(x)$. If, for some generator $g \in S$, $v$ is a maximum of $L_{g}$, then without loss of generality, pick $v$ among the maximum points of $L_{g}$ with the highest value of $-L_{d+1}$ (maximize $\sum_{g \in S} L_{g}(v)$). In addition, since the rooted stabilizer $z$ is supported only on the vertices in $\textbf{Str}_{0+}(v,x)$, the addition cannot introduce a vertex with a higher $L_{g}$ value than the maximal $L_{g}$ value in $\textbf{Str}_{0+}(v,x)$ into $x$.

  Now consider the skeleton system where the directions are flipped, and $\rotfs{}$ is set to be the face containing the vertices that are reached by the selection of reverse steps from $S^{\prime}$ (i.e., $g^{-1}v \in \rotfs{}$). In that system, we can apply \Cref{claim:agreement} (in the reversed direction) and get that there is a stabilizer $z$ such that adding it to $x$ excludes $v$ from $x$. By the same argument as above, no vertex with a higher $L_{g^{\prime}}$-value (although $L_{d+1}$ can increase, that's fine) than what it already has in $\textbf{Str}_{0-}(v,x)$ is introduced into $x$. Repeating the process excludes all the vertices with the maximal value of $L_{g}$ in $x$, so eventually, we end up with $x+z$, such that $L_{g}(x+z) < L_{g}(x)$.

  
We keep excluding vertices until we get the desired equality.
\end{proof}

\begin{claim}
  The function  
  \begin{equation*}
    \begin{split}
      f(x) \colonequals \min_{z \in \mathcal{C}_{Z}^{\perp}} \Size{ x + z }  
    \end{split}
  \end{equation*} 
  is a function that respects inclusion (\Cref{def:respinc}). Here, the union/intersection of chains is regarding their $d^{\prime}$-dimensional faces (bits).
\end{claim}
\begin{proof}
  We will show that all the three conditions hold. 
\item If $x$ is the zero vector, then the bits supporting it are the empty set, and $f(x) = \Size{ \mathbf{0} } = 0$. For the empty set, $f(\emptyset) = 0$.
\item Consider two chains $x,y$ that has an intersection, we would like to show that $f(x\cup y) \le f(x) + f(y)$, notice that the union is a set-union (regarding the faces that support the chain) and not vectors addition. It's easy to see that the faces of $(x\cup y) + z$ are contained in the faces of  $(x+z)\cup y$. Thus we have: 
  \begin{equation*}
    \begin{split}
      \min_{z \in \mathcal{C}_{Z}^{\perp}} \Size{(x \cup y) + z} &= \min_{z,w \in \mathcal{C}_{Z}^{\perp}} \Size{(x \cup y) + z+ w} \\ 
      & \le \min_{z,w \in \mathcal{C}_{Z}^{\perp}} \Size{ (x +z) \cup (y + w)}  \\
      & \le \Size{ (x +z) \cup (y + w)} \text{ such } \substack{ z = \arg \min \Size{ x + z } \\  w = \arg \min \Size{ y + w } } \\ 
    & \le \min_{z \in \mathcal{C}_{Z}^{\perp}} \Size{ x +z } + \min_{w \in \mathcal{C}_{Z}^{\perp}}   \Size{y + w}  \\
    \end{split}
  \end{equation*}
  Now, if $x+z$ and $y+w$ share a vertex then, $\Size{ (x +z) \cup (y + w)} \le \Size{ x + z} + \Size{ y+ w}$. 
    \item There is a constant $c$ such that the function $f$ on any singleton (a subset containing a single vertex) is bounded by $c$.
\end{proof}
